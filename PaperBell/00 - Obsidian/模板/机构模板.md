<%*
// 取消提示框时 tp.system.prompt 返回 null，统一收敛成空串
const ask = async (q) => (await tp.system.prompt(q, "") || "").trim();
const lat = await ask("请输入纬度（不知道可留空）");
const lon = await ask("请输入经度（不知道可留空）");
const website = await ask("请输入机构网站");
const abbr = await ask("请输入机构简称");
const alias = await ask("请输入机构别名（如中文称呼）");
const logo = await ask("请输入Logo图片路径");
// 坐标必须成对且为数字才写入。`location: [, ]` 会让整块 frontmatter 解析失败，
// 连 name / tags 一起丢掉，所以留空时整个键都不出现。
const hasGeo = lat !== "" && lon !== "" && !isNaN(Number(lat)) && !isNaN(Number(lon));
if (abbr) tR += `abbr: ${abbr}\n`;
if (alias) tR += `aliases:\n  - ${alias}\n`;
if (website) tR += `website: ${website}\n`;
if (hasGeo) tR += `location:\n  - ${lat}\n  - ${lon}\n`;
if (logo) tR += `logo: ${logo}\n`;
-%>
name: <% tp.file.title %>
tags:
- institute
---

# <% website ? `[${tp.file.title}](${website})` : tp.file.title %>

## Overview
<%*
if (hasGeo) {
	const view = { name: "Default", mapZoom: 8, centerLat: Number(lat), centerLng: Number(lon), query: "", chosenMapSource: 0 };
	tR += "\n```mapview\n" + JSON.stringify(view) + "\n```\n";
} else {
	tR += "\n> [!tip] 补上 `location` 坐标后，可以在这里放一个 `mapview` 代码块显示机构位置。\n";
}
-%>
